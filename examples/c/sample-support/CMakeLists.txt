cmake_minimum_required(VERSION 3.19)
project(ovxr-sample-support LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/sample_support_headers.cmake")
if(NOT SAMPLE_OPENXR_INCLUDE_DIR OR NOT SAMPLE_VULKAN_INCLUDE_DIR)
    message(FATAL_ERROR "Set SAMPLE_OPENXR_INCLUDE_DIR (OpenXR 1.1.53+) and SAMPLE_VULKAN_INCLUDE_DIR "
        "to their SDK include directories for this support-dependent sample.")
endif()
add_library(ovxr-sample-support SHARED input.cpp mirror.cpp panel.cpp observer.cpp projection.cpp)
target_link_libraries(ovxr-sample-support PRIVATE ovstage::ovstage ovrtx::ovrtx)
target_compile_features(ovxr-sample-support PRIVATE cxx_std_17)
target_compile_definitions(ovxr-sample-support PRIVATE SAMPLE_SUPPORT_BUILD XR_NO_PROTOTYPES)
target_include_directories(ovxr-sample-support PRIVATE "${SAMPLE_OPENXR_INCLUDE_DIR}" "${SAMPLE_VULKAN_INCLUDE_DIR}")
target_link_libraries(ovxr-sample-support PRIVATE ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(ovxr-sample-support PRIVATE user32 gdi32)
endif()
